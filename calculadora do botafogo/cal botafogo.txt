    <script>
        const display = document.getElementById('display');
        let expr = '';

        function update(){ display.textContent = expr || '0'; }

        function safeEval(s){
            // substitui sinais visuais por operadores JS
            try{
                const sanitized = s.replace(/×/g,'*').replace(/÷/g,'/').replace(/−/g,'-');
                // Evita caracteres inesperados
                if(/[^0-9.+\-*/()% ]/.test(sanitized)) return 'Erro';
                // eslint-disable-next-line no-eval
                const res = eval(sanitized);
                return (res === Infinity || Number.isNaN(res)) ? 'Erro' : String(res);
            }catch(e){ return 'Erro'; }
        }

        document.querySelectorAll('button').forEach(btn => {
            btn.addEventListener('click', () => {
                const v = btn.dataset.val;
                if(v === 'C'){ expr = ''; update(); return; }
                if(v === '←'){ expr = expr.slice(0,-1); update(); return; }
                if(v === '='){ expr = safeEval(expr); update(); return; }
                expr += v;
                update();
            });
        });
    </script>
